Datasets

We use left knee flexion-extension data from two open source datasets.

Parkinson’s Disease

The first dataset is from individuals living with Parkinson’s Disease. This is obtained from:

Daniel Boari (2021). A dataset of overground walking full-body kinematics and kinetics in individuals with Parkinson’s disease. figshare. Dataset. https://doi.org/10.6084/m9.figshare.14896881.v4

Healthy Dataset

The second dataset is from healthy individuals. This is obtained from:

Claudiane Fukuchi, Reginaldo Fukuchi, and Marcos Duarte. A public data set of overground and treadmill walking kinematics and kinetics of healthy individuals, 2018. URL https://figshare.com/articles/dataset/A_public_data_set_of_overground_and_treadmill_walking_kinematics_and_kinetics_of_healthy_individuals/5722711

Please see “01_combine_datasets.R” to see how left knee flexion-extension data is extracted from the datasets.

Assessing sample size calculations

We compute the empirical power of a paired two-sided Wilcoxon signed-rank test (with \(\alpha = 0.05\)) to detect a non-zero change in each score (clinical outcomes, MFPCA scores, mean of swing/stance peak) as sample size increases. Power is estimated based on 1000 bootstrap resamples drawn with replacement from the PD dataset.

all_scores = data.frame(FPC1_on=predicted_scores_on$L1_PC1,
                        FPC1_off=predicted_scores_off$L1_PC1, 
                        FPC2_on=predicted_scores_on$L1_PC2, 
                        FPC2_off = predicted_scores_off$L1_PC2, 
                        FPC3_on=predicted_scores_on$L1_PC3, 
                        FPC3_off = predicted_scores_off$L1_PC3, 
                        UPDRS_II_on=as.numeric(diagnosis_wide$UPDRS_II_ON),
                        UPDRS_II_off=as.numeric(diagnosis_wide$UPDRS_II_OFF), 
                        UPDRS_III_on=as.numeric(diagnosis_wide$UPDRS_III_ON), 
                        UPDRS_III_off= as.numeric(diagnosis_wide$UPDRS_III_OFF), 
                        mean_swing_peak_on = peak_data[peak_data$Medication=="on", "mean_swing_peak"]$mean_swing_peak, 
                        mean_swing_peak_off = peak_data[peak_data$Medication=="off", "mean_swing_peak"]$mean_swing_peak, 
                        mean_stance_peak_on = peak_data[peak_data$Medication=="on", "mean_stance_peak"]$mean_stance_peak, 
                        mean_stance_peak_off = peak_data[peak_data$Medication=="off", "mean_stance_peak"]$mean_stance_peak)


library(coin)

set.seed(112026)

all_results <- c()

for (i in 1:1000){
  for (sample_size in c(8:22)){
    ids <- sample(1:22, sample_size, replace=TRUE)
    
    subset<-all_scores[ids,]
    
    FPC1 = pvalue(wilcoxsign_test(FPC1_on ~ FPC1_off, data=subset, distribution = "exact")) <0.05
    FPC2 = pvalue(wilcoxsign_test(FPC2_on ~ FPC2_off, data=subset, distribution = "exact")) <0.05
    FPC3 = pvalue(wilcoxsign_test(FPC3_on ~ FPC3_off, data=subset, distribution = "exact")) <0.05
    UPDRS_II = pvalue(wilcoxsign_test(UPDRS_II_on ~ UPDRS_II_off, data=subset, distribution = "exact")) <0.05
    UPDRS_III  = pvalue(wilcoxsign_test(UPDRS_III_on ~ UPDRS_III_off, data=subset, distribution = "exact")) <0.05
    mean_swing_peak = pvalue(wilcoxsign_test(mean_swing_peak_on ~ mean_swing_peak_off, data=subset, distribution = "exact")) <0.05
    mean_stance_peak = pvalue(wilcoxsign_test(mean_stance_peak_on ~ mean_stance_peak_off, data=subset, distribution = "exact")) <0.05
    
    
    result <- cbind(sim=i, sample_size=sample_size, FPC1=FPC1, FPC2=FPC2, FPC3=FPC3, 
                    UPDRS_II=UPDRS_II, UPDRS_III=UPDRS_III, mean_stance_peak=mean_stance_peak, 
                    mean_swing_peak=mean_swing_peak )
    all_results <- rbind(all_results, result)
  }
}

#plot results 

B=1000
sample_size_power <- all_results %>%
  pivot_longer(
    cols = c(FPC1, FPC2, FPC3, UPDRS_II, UPDRS_III, mean_stance_peak, mean_swing_peak),
    names_to = "Outcome",
    values_to = "significant"
  ) %>%
  mutate(
    Outcome = case_match(
      Outcome,
      "FPC1"      ~ "MFPCA Score 1",
      "FPC2"      ~ "MFPCA Score 2",
      "FPC3"      ~ "MFPCA Score 3",
      "UPDRS_II"  ~ "MDS UPDRS II",
      "UPDRS_III" ~ "MDS UPDRS III",
      "mean_stance_peak" ~ "Mean Stance Peak",
      "mean_swing_peak" ~ "Mean Swing Peak",
      .default    = Outcome
    )
  ) %>%
  group_by(sample_size, Outcome) %>%
  summarise(
    B     = sum(!is.na(significant)),
    power = mean(significant, na.rm = TRUE),
    mcse  = sd(significant, na.rm = TRUE) / sqrt(B),
    ymin  = pmax(0, power - 1.96 * mcse),
    ymax  = pmin(1, power + 1.96 * mcse),
    .groups = "drop"
  )

power_plot <- ggplot(sample_size_power, aes(x = sample_size, y = power, color = Outcome, fill = Outcome)) +
